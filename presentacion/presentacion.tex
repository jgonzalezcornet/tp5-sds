\section{Introducci\'on}

\begin{frame}{Sistema real}
    \small
    \begin{itemize}
      \item Red de neuronas simplificadas con actividad oscilatoria.
      \item Cada neurona se modela por una fase $\theta_i(t)$ y una frecuencia natural $\omega_i$.
      \item El acoplamiento entre neuronas puede inducir sincronizaci\'on colectiva.
    \end{itemize}
\end{frame}

\begin{frame}{Modelo matem\'atico}
  \small
  \textbf{Dinámica temporal de la i-ésima neurona:}
  \begin{equation*}
    \frac{d\theta_i}{dt} = \omega_i + K \sum_j A_{ij} \sin(\theta_j - \theta_i)
  \end{equation*}

  \begin{columns}[T,onlytextwidth]
    \column{0.48\textwidth}
    \begin{itemize}
      \item $\theta_i$: fase de la neurona $i$.
      \item $\omega_i$: frecuencia natural de la neurona $i$.
      \item $K$: intensidad de acoplamiento global.
    \end{itemize}

    \column{0.48\textwidth}
    \begin{itemize}
      \item $A_{ij}$: matriz de conectividad de la red.
    \end{itemize}
  \end{columns}

  \vspace{0.4cm}
  \textbf{Especificaci\'on del modelo}
  \begin{itemize}
    \item $\omega_i \sim \mathcal{N}(1, 0.1^2)$
    \item $\theta_i \sim \mathcal{U}[0, 2\pi)$
    % \item La topolog\'ia queda codificada en $A_{ij}$.
  \end{itemize}
\end{frame}

\begin{frame}{Algoritmo implementado}
  \begin{algorithm}[H]
    \small
    \caption{Bucle principal del motor}
    \begin{algorithmic}[1]
      \State Muestrear $\omega_i$ y $\theta_i(0)$; construir la red.
      \For{cada paso de tiempo}
        \State Avanzar $\theta$ con un paso RK4.

\begin{frame}{Par\'ametros del estudio}
  \begin{columns}[T,onlytextwidth]
    \hspace{0.5cm}
    \column{0.4\textwidth}
    \small
    \textbf{Par\'ametros fijos}
    \begin{itemize}
      \item $N = 600$
      \item $dt = 10^{-3}$
    \end{itemize}

    \column{0.6\textwidth}
    \small
    \textbf{Variables}
    \begin{itemize}
      \item Completa: $K \in [0,1]$
      \item Aleatoria: $p \in [10^{-4},10^{-1}]$
      \item Anillo: $v \in \{1,\dots,10\}$
    \end{itemize}
  \end{columns}

  \vspace{0.35cm}
  \hspace{0.5cm}
  \small
  \textbf{Protocolo}
  \begin{itemize}
    \item 12 seeds por simulación
    \item $t_{\mathrm{sim}}$: 50 s (completa, aleatoria), 1500 s (anillo)
  \end{itemize}
\end{frame}

\begin{frame}{Observables definidos}
  \small
  \textbf{Par\'ametro de orden}
  \begin{equation*}
    r(t) = \left| \frac{1}{N} \sum_j [\cos(\theta_j), \sin(\theta_j)] \right|
  \end{equation*}

  \textbf{Valor estacionario}
  \begin{equation*}
    r_{\infty} = \frac{1}{t_f - t_{\mathrm{est}}} \int_{t_{\mathrm{est}}}^{t_f} r(t)\,dt
  \end{equation*}
  {\footnotesize donde $t_{\mathrm{est}}$ es el 80\% de $t_f$}
  
  \vspace{0.4cm}

  \textbf{Tiempo de sincronizaci\'on}
  \begin{equation*}
    \tau = \min \left\{ t : r(t) \ge 0.95\, r_{\infty} \right\}
  \end{equation*}

  {\footnotesize $\tau$ es el primer instante en que alcanza el $95\%$ de $r_{\infty}$.}
\end{frame}
