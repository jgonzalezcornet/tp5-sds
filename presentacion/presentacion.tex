\begin{frame}{Modelo matem\'atico}
  \small
  \textbf{Dinámica temporal de la i-ésima neurona:}
  \begin{equation*}
    \frac{d\theta_i}{dt} = \omega_i + K \sum_j A_{ij} \sin(\theta_j - \theta_i)
  \end{equation*}

  \vspace{0.25cm}

  \begin{columns}[T,onlytextwidth]
    \column{0.52\textwidth}
    \begin{itemize}
      \item $\theta_i$: fase de la neurona $i$.
      \item $\omega_i$: frecuencia natural de la neurona $i$.
      \item $A_{ij}$: matriz de conectividad de la red.
      \item $K$: intensidad de acoplamiento global.
    \end{itemize}

    \column{0.44\textwidth}
    \textbf{Especificaci\'on del modelo}
    \begin{itemize}
      \item $\omega_i \sim \mathcal{N}(1, 0.1^2)$
      \item $\theta_i(0) \sim \mathcal{U}[0, 2\pi)$
    \end{itemize}
  \end{columns}
\end{frame}

\begin{frame}{Par\'ametros del estudio}
  \begin{columns}[T,onlytextwidth]
    \hspace{0.5cm}
    \column{0.4\textwidth}
    \small
    \textbf{Par\'ametros fijos}
    \begin{itemize}
      \item $N = 600$
      \item $dt = 10^{-3}$
    \end{itemize}

    \column{0.6\textwidth}
    \small
    \textbf{Variables}
    \begin{itemize}
      \item Completa: $K \in [0,1]$
      \item Aleatoria: $p \in [10^{-4},10^{-1}]$
      \item Anillo: $v \in \{1,\dots,10\}$
    \end{itemize}
  \end{columns}

  \vspace{0.35cm}
  \hspace{0.5cm}
  \small
  \textbf{Protocolo}
  \begin{itemize}
    \item Realizaciones: 12 (completa, aleatoria), 50 (anillo)
    \item $t_{\mathrm{sim}}$: 50 s (completa), 100 s (aleatoria), 1500 s (anillo)
  \end{itemize}
\end{frame}

\begin{frame}{Observables definidos}
  \small
  \textbf{Par\'ametro de orden}
  \begin{equation*}
    r(t) = \left| \frac{1}{N} \sum_j [\cos(\theta_j), \sin(\theta_j)] \right|
  \end{equation*}

  \textbf{Tiempo de sincronizaci\'on (llegada al estacionario)}
  \begin{equation*}
    \tau = \min\Big\{ t : \max_{s \ge t} r(s) - \min_{s \ge t} r(s) \le 0.04 \Big\}
  \end{equation*}
  {\footnotesize $\tau$: primer instante a partir del cual $r(t)$ deja de variar ($\pm 0.02$).}

  \vspace{0.4cm}

  \textbf{Valor estacionario}
  \begin{equation*}
    r_{\infty} = \frac{1}{M} \sum_{t_k \ge \tau} r(t_k)
  \end{equation*}
  {\footnotesize promedio de $r$ sobre la ventana ya estable.}
  
\end{frame}
